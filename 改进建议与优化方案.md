#### 1. 预计算梯度纹理

**当前问题：**
```glsl
// 每个采样点需要6次额外的纹理采样来计算梯度
vec3 computeSimpleGradient(vec3 pos) {
    float dx_pos = texture(volume, pos + vec3(eps,0,0)).r;
    float dx_neg = texture(volume, pos - vec3(eps,0,0)).r;
    float dy_pos = texture(volume, pos + vec3(0,eps,0)).r;
    float dy_neg = texture(volume, pos - vec3(0,eps,0)).r;
    float dz_pos = texture(volume, pos + vec3(0,0,eps)).r;
    float dz_neg = texture(volume, pos - vec3(0,0,eps)).r;
    // 总共6次采样，非常昂贵！
}
```

**改进方案：**
```cpp
// 在CPU端预计算梯度，存储为RGBA纹理
struct VolumeWithGradient {
    GLuint textureID;  // RGBA32F格式
    // R通道: 密度
    // GBA通道: 梯度xyz
};

VolumeWithGradient createVolumeWithPrecomputedGradient(
    const VolumeBuildResult& volumeData) 
{
    int width = volumeData.width;
    int height = volumeData.height;
    int depth = volumeData.depth;
    
    std::vector<glm::vec4> volumeRGBA(width * height * depth);
    
    for (int z = 0; z < depth; z++) {
        for (int y = 0; y < height; y++) {
            for (int x = 0; x < width; x++) {
                int idx = x + y * width + z * width * height;
                
                // 密度值
                float density = getDensity(volumeData, x, y, z);
                
                // 计算梯度（Sobel算子）
                glm::vec3 gradient = computeGradientCPU(
                    volumeData, x, y, z);
                
                volumeRGBA[idx] = glm::vec4(density, gradient);
            }
        }
    }
    
    // 创建RGBA纹理
    GLuint texID;
    glGenTextures(1, &texID);
    glBindTexture(GL_TEXTURE_3D, texID);
    glTexImage3D(GL_TEXTURE_3D, 0, GL_RGBA32F,
                 width, height, depth,
                 0, GL_RGBA, GL_FLOAT, volumeRGBA.data());
    // ...设置纹理参数
    
    return {texID};
}
```

**着色器简化：**
```glsl
// 从1+6次采样减少到1次采样
vec4 sample = texture(volumeWithGradient, samplePos);
float density = sample.r;
vec3 normal = normalize(sample.gba);  // 直接获取预计算梯度
```

**性能提升：** 约 **6倍** 的采样速度提升

#### 2. 八叉树加速结构（Octree Acceleration）

**原理：** 将体数据分割成层次化的块，跳过完全透明的区域

**数据结构：**
```cpp
struct OctreeNode {
    glm::vec3 minBound;
    glm::vec3 maxBound;
    float minDensity;   // 该区域的最小密度
    float maxDensity;   // 该区域的最大密度
    bool isLeaf;
    std::array<OctreeNode*, 8> children;
};

class VolumeOctree {
public:
    void build(const VolumeBuildResult& volume, int maxDepth = 5);
    
    // 生成用于GPU的纹理表示
    GLuint createOctreeTexture();
    
private:
    OctreeNode* root;
};
```

**GPU实现（使用3D纹理存储八叉树）：**
```glsl
uniform sampler3D octreeTexture;  // 每个节点存储 (minDensity, maxDensity)

bool skipEmptyRegion(vec3 rayPos, vec3 rayDir, 
                     float windowLevel, float windowWidth,
                     out float skipDistance) 
{
    // 查询八叉树，判断当前区域是否可以跳过
    vec2 densityRange = texture(octreeTexture, rayPos).rg;
    
    float windowMin = windowLevel - windowWidth / 2.0;
    float windowMax = windowLevel + windowWidth / 2.0;
    
    // 如果整个区域的最大密度都低于窗口最小值，可以跳过
    if (densityRange.y < windowMin) {
        skipDistance = computeSkipDistance(rayPos, rayDir);
        return true;
    }
    
    return false;
}

// 在主循环中使用
for (int i = 0; i < numSteps; ++i) {
    float skipDist;
    if (skipEmptyRegion(samplePos, rayDir, windowLevel, windowWidth, skipDist)) {
        samplePos += rayDir * skipDist;  // 跳过空区域
        continue;
    }
    
    // 正常采样...
}
```

**性能提升：** 对于包含大量空区域的数据（如CT扫描），可提升 **2-5倍**

#### 3. 自适应采样率

**原理：** 根据密度梯度动态调整步长

```glsl
float adaptiveStepSize(vec3 samplePos, vec3 normal, float basestepSize) 
{
    // 梯度大的地方（边缘）用小步长，梯度小的地方用大步长
    float gradientMagnitude = length(normal);
    
    // 梯度范围 [0, 1]，映射到步长倍数 [0.5, 3.0]
    float stepMultiplier = mix(3.0, 0.5, gradientMagnitude);
    
    return baseStepSize * stepMultiplier;
}

// 在主循环中
float currentStepSize = stepSize;
for (float t = 0.0; t < rayLength; t += currentStepSize) 
{
    vec3 samplePos = entryPoint + rayDir * t;
    vec4 sample = texture(volumeWithGradient, samplePos);
    vec3 normal = normalize(sample.gba);
    
    // 动态调整下一步的步长
    currentStepSize = adaptiveStepSize(samplePos, normal, stepSize);
    
    // ... 正常采样和混合
}
```

**效果：** 在保持边缘质量的同时，提升 **30-50%** 性能

#### 4. 体积阴影（Volumetric Shadows）

**效果：** 使内部结构更有层次感

```glsl
float computeVolumeShadow(vec3 position, vec3 lightDir) 
{
    const int shadowSteps = 8;  // 阴影采样步数
    const float shadowStepSize = 0.05;
    
    float shadowFactor = 1.0;
    vec3 shadowPos = position;
    
    // 沿光线方向向光源采样
    for (int i = 0; i < shadowSteps; i++) {
        shadowPos += lightDir * shadowStepSize;
        
        // 边界检查
        if (any(lessThan(shadowPos, vec3(0.0))) || 
            any(greaterThan(shadowPos, vec3(1.0)))) {
            break;
        }
        
        // 采样密度
        float density = texture(volume, shadowPos).r;
        float windowedDensity = applyWindow(density, windowLevel, windowWidth);
        
        // 累积遮挡
        shadowFactor *= (1.0 - windowedDensity * 0.1);
        
        if (shadowFactor < 0.01) break;  // 早期终止
    }
    
    return shadowFactor;
}

// 在光照计算中使用
vec3 litColor = applyPhongLighting(...);
float shadow = computeVolumeShadow(samplePos, lightDir);
litColor *= shadow;  // 应用阴影
```

**性能开销：** 每个采样点额外8次纹理采样（可配置）

#### 5. 环境光遮蔽（Ambient Occlusion）

**效果：** 增强深度感和接触感

```glsl
float computeAmbientOcclusion(vec3 position, vec3 normal) 
{
    const int aoSamples = 8;
    const float aoRadius = 0.05;
    
    float occlusion = 0.0;
    
    for (int i = 0; i < aoSamples; i++) {
        // 生成半球采样方向
        vec3 sampleDir = generateHemisphereSample(normal, i, aoSamples);
        vec3 samplePos = position + sampleDir * aoRadius;
        
        // 采样密度
        float density = texture(volume, samplePos).r;
        float windowedDensity = applyWindow(density, windowLevel, windowWidth);
        
        // 累积遮蔽
        occlusion += windowedDensity;
    }
    
    occlusion /= float(aoSamples);
    return 1.0 - clamp(occlusion, 0.0, 1.0);
}

// 应用到环境光
vec3 ambient = ambientStrength * color * computeAmbientOcclusion(samplePos, normal);
```

#### 8. 裁剪平面（Clipping Planes）

**用途：** 切开体数据观察内部结构

```cpp
// C++端定义裁剪平面
struct ClippingPlane {
    glm::vec3 point;   // 平面上的一点
    glm::vec3 normal;  // 平面法向量
    bool enabled;
};

std::vector<ClippingPlane> clippingPlanes;
```

```glsl
// 着色器中检查
uniform vec4 clippingPlanes[6];  // 最多6个裁剪平面 (xyz=法向, w=距离)
uniform int numClippingPlanes;

bool isClipped(vec3 position) 
{
    for (int i = 0; i < numClippingPlanes; i++) {
        vec3 normal = clippingPlanes[i].xyz;
        float distance = clippingPlanes[i].w;
        
        // 点到平面距离
        float d = dot(position, normal) - distance;
        
        if (d < 0.0) {
            return true;  // 被裁剪
        }
    }
    return false;
}

// 在采样循环中
if (isClipped(samplePos)) {
    discard;  // 或者 continue
}
```

```cpp
std::string Shader::loadShaderWithIncludes(const std::string& filepath) 
{
    std::string source = readFile(filepath);
    
    // 查找 #include 指令
    size_t pos = 0;
    while ((pos = source.find("#include", pos)) != std::string::npos) {
        size_t start = source.find("\"", pos) + 1;
        size_t end = source.find("\"", start);
        std::string includePath = source.substr(start, end - start);
        
        // 递归加载包含文件
        std::string includedSource = loadShaderWithIncludes(
            getDirectory(filepath) + "/" + includePath);
        
        // 替换 #include 行
        source.replace(pos, end - pos + 1, includedSource);
    }
    
    return source;
}
```

```cpp
class PerformanceMonitor {
public:
    void beginFrame() {
        frameStartTime = glfwGetTime();
    }
    
    void endFrame() {
        double frameTime = glfwGetTime() - frameStartTime;
        frameTimes.push_back(frameTime);
        
        if (frameTimes.size() > 60) {
            frameTimes.erase(frameTimes.begin());
        }
    }
    
    float getAverageFPS() {
        if (frameTimes.empty()) return 0.0f;
        
        double sum = 0.0;
        for (double t : frameTimes) sum += t;
        double avgTime = sum / frameTimes.size();
        
        return 1.0f / avgTime;
    }
    
    float getFrameTimeMS() {
        return frameTimes.empty() ? 0.0f : frameTimes.back() * 1000.0f;
    }
    
private:
    double frameStartTime;
    std::vector<double> frameTimes;
};
```

```cpp
class ScreenCapture {
public:
    void captureScreenshot(const std::string& filename, int width, int height) {
        std::vector<unsigned char> pixels(width * height * 3);
        
        // 读取帧缓冲
        glReadPixels(0, 0, width, height, GL_RGB, GL_UNSIGNED_BYTE, pixels.data());
        
        // 翻转Y轴（OpenGL坐标系）
        std::vector<unsigned char> flipped(width * height * 3);
        for (int y = 0; y < height; y++) {
            memcpy(&flipped[y * width * 3], 
                   &pixels[(height - 1 - y) * width * 3], 
                   width * 3);
        }
        
        // 保存为PNG（使用stb_image_write）
        stbi_write_png(filename.c_str(), width, height, 3, flipped.data(), width * 3);
    }
    
    void startRecording(int width, int height, int fps = 30) {
        isRecording = true;
        recordedFrames.clear();
        recordWidth = width;
        recordHeight = height;
        recordFPS = fps;
    }
    
    void captureFrame(int width, int height) {
        if (!isRecording) return;
        
        std::vector<unsigned char> pixels(width * height * 3);
        glReadPixels(0, 0, width, height, GL_RGB, GL_UNSIGNED_BYTE, pixels.data());
        recordedFrames.push_back(pixels);
    }
    
    void stopRecording(const std::string& filename) {
        if (!isRecording) return;
        isRecording = false;
        
        // 保存为视频文件（需要ffmpeg或其他视频编码库）
        // ... 实现视频编码
    }
    
private:
    bool isRecording = false;
    std::vector<std::vector<unsigned char>> recordedFrames;
    int recordWidth, recordHeight, recordFPS;
};
```
